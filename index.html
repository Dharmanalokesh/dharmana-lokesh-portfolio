hotel_cancellation_analysis.sql 
SELECT 
    r.room_type,
    COUNT(b.booking_id) AS total_bookings,
    SUM(CASE WHEN b.status = 'Canceled' 
             THEN 1 ELSE 0 END) AS cancellations,
    ROUND(SUM(CASE WHEN b.status = 'Canceled' 
                   THEN 1 ELSE 0 END) * 100.0 / COUNT(*), 2) AS cancel_rate_pct
FROM bookings b
JOIN rooms r ON b.room_id = r.room_id
GROUP BY r.room_type
HAVING COUNT(b.booking_id) >= 50
ORDER BY cancel_rate_pct DESC;
